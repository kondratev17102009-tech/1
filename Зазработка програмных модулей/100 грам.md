ЗАДАНИЕ 2. Диапазон: чётные, нечётные или кратные 7
``` C++
#include <iostream> // Подключаем ввод и вывод
using namespace std; // Подключаем стандартное пространство имён

int main() { // Начало программы
    int a, b; // Две границы диапазона
    cout << "Введите две границы: "; // Просим границы
    cin >> a >> b; // Считываем границы

    if (a > b) { // Если границы введены в обратном порядке
        int temp = a; // Временно сохраняем первую границу
        a = b; // На место первой ставим вторую
        b = temp; // На место второй ставим сохранённую первую
    } // Конец перестановки

    int mode; // Выбор вида чисел
    cout << "1 - четные, 2 - нечетные, 3 - кратные 7: "; // Показываем меню
    cin >> mode; // Считываем выбор

    int i = a; // Начинаем с меньшей границы
    while (i <= b) { // Перебираем числа до большей границы
        if (mode == 1 && i % 2 == 0) { // Если выбраны чётные и число делится на 2
            cout << i << " "; // Выводим чётное число
        } else if (mode == 2 && i % 2 != 0) { // Если выбраны нечётные
            cout << i << " "; // Выводим нечётное число
        } else if (mode == 3 && i % 7 == 0) { // Если число делится на 7 без остатка
            cout << i << " "; // Выводим число
        } // Конец проверок
        i++; // Переходим к следующему числу
    } // Конец цикла

    cout << endl; // Перенос строки
    return 0; // Завершаем программу
}
```

ЗАДАНИЕ 5. Игра «Угадай число»

``` C++

#include <iostream> // Подключаем ввод и вывод
#include <cstdlib> // Подключаем rand и srand
#include <ctime> // Подключаем time
using namespace std; // Подключаем стандартное пространство имён

int main() { // Начало программы
    srand(static_cast<unsigned int>(time(0))); // Задаём случайное начальное значение
    int secret = rand() % 500 + 1; // Загадываем число от 1 до 500
    int guess = -1; // Переменная для ответа игрока
    int attempts = 0; // Счётчик попыток
    time_t startTime = time(0); // Запоминаем время начала игры

    cout << "Угадайте число от 1 до 500. Введите 0 для выхода." << endl; // Правила игры
    while (guess != secret && guess != 0) { // Играем, пока число не угадано и не введён 0
        cout << "Ваш вариант: "; // Просим вариант
        cin >> guess; // Считываем вариант

        if (guess == 0) { // Если игрок решил выйти
            break; // Выходим из цикла
        } // Конец проверки выхода

        attempts++; // Считаем попытку
        if (guess < secret) { // Если вариант меньше загаданного
            cout << "Загаданное число больше." << endl; // Подсказка
        } else if (guess > secret) { // Если вариант больше загаданного
            cout << "Загаданное число меньше." << endl; // Подсказка
        } else { // Иначе число угадано
            cout << "Правильно!" << endl; // Сообщаем об успехе
        } // Конец сравнения
    } // Конец игры

    time_t endTime = time(0); // Запоминаем время окончания
    if (guess == secret) { // Если игрок угадал число
        cout << "Попыток: " << attempts << endl; // Выводим количество попыток
        cout << "Время: " << endTime - startTime << " секунд" << endl; // Выводим длительность
    } else { // Если игрок вышел
        cout << "Игра завершена досрочно." << endl; // Сообщаем о выходе
    } // Конец итоговой проверки

    return 0; // Завершаем программу
}

```

``` C++
    cout << "1. RUB -> USD" << endl; // Пункт меню
    cout << "2. USD -> RUB" << endl; // Пункт меню
    cout << "3. RUB -> EUR" << endl; // Пункт меню
    cout << "4. EUR -> RUB" << endl; // Пункт меню
    cout << "5. RUB -> CNY" << endl; // Пункт меню
    cout << "6. CNY -> RUB" << endl; // Пункт меню

    cout << "8. GBP -> RUB" << endl; // Пункт меню
    cout << "0. Выход" << endl; // Пункт выхода

    choice = -1; // Задаём значение, чтобы войти в цикл

        cout << "Выберите действие: "; // Просим выбрать пункт
        cin >> choice; // Считываем пункт
        if (choice == 0) { // Если выбрали выход
            break; // Заканчиваем цикл
        } // Конец проверки выхода

        cout << "Введите сумму: "; // Просим сумму
        cin >> amount; // Считываем сумму
        if (amount < 0) { // Проверяем сумму
            cout << "Сумма не может быть отрицательной." << endl; // Сообщаем об ошибке
            continue; // Возвращаемся к началу цикла
        } // Конец проверки суммы

        result = 0; // Изначально результат нулевой
        if (choice == 1) result = amount / USD; // Переводим рубли в доллары
        else if (choice == 2) result = amount * USD; // Переводим доллары в рубли
        else if (choice == 3) result = amount / EUR; // Переводим рубли в евро
        else if (choice == 4) result = amount * EUR; // Переводим евро в рубли
        else if (choice == 5) result = amount / CNY; // Переводим рубли в юани
        else if (choice == 6) result = amount * CNY; // Переводим юани в рубли
        else if (choice == 7) result = amount / GBP; // Переводим рубли в фунты
        else if (choice == 8) result = amount * GBP; // Переводим фунты в рубли
        else { // Если номер не относится к меню
            cout << "Нет такого пункта меню." << endl; // Сообщаем об ошибке
            continue; // Возвращаемся к меню
        } // Конец выбора курса

        cout << "Результат = " << result << endl; // Выводим результат
    } // Конец цикла меню

    cout << "Конвертер закрыт." << endl; // Сообщаем о завершении
    return 0; // Завершаем программу
```

ЗАДАНИЕ 8. x в степени y через цикл, без pow() и ^

``` C++
int main() { // Начало программы
    double x; // Основание степени
    int y; // Показатель степени
    cout << "Введите x и целое y: "; // Просим числа
    cin >> x >> y; // Считываем числа

    double result = 1; // Начинаем с единицы
    int count = 0; // Счётчик умножений
    int power = y; // Копируем показатель
    if (power < 0) { // Если степень отрицательная
        power = -power; // Берём её положительное значение
    } // Конец проверки

    while (count < power) { // Повторяем нужное количество раз
        result *= x; // Умножаем результат на x
        count++; // Считаем умножение
    } // Конец цикла

    if (y < 0) { // Если исходная степень была отрицательной
        if (x == 0) { // Ноль нельзя возводить в отрицательную степень
            cout << "Ошибка: 0 в отрицательной степени не определён." << endl; // Выводим ошибку
            return 0; // Завершаем программу
        } // Конец проверки нуля
        result = 1 / result; // Для отрицательной степени берём обратное число
    } // Конец обработки отрицательной степени

    cout << "Результат = " << result << endl; // Выводим ответ
```

ЗАДАНИЕ 12. Проверка знания таблицы умножения

``` C++

#include <iostream> // Подключаем ввод и вывод
#include <cstdlib> // Подключаем rand и srand
#include <ctime> // Подключаем time
using namespace std; // Подключаем стандартное пространство имён

int main() { // Начало программы
    srand(static_cast<unsigned int>(time(0))); // Подготавливаем генератор случайных чисел
    int level; // Уровень сложности
    cout << "Выберите уровень: 1 - лёгкий (5 вопросов), 2 - средний (10), 3 - сложный (15): "; // Показываем уровни
    cin >> level; // Считываем уровень

    int questions = 5; // По умолчанию задаём 5 вопросов
    int maxNumber = 5; // По умолчанию числа до 5
    if (level == 2) { // Если выбран средний уровень
        questions = 10; // Задаём 10 вопросов
        maxNumber = 10; // Числа до 10
    } else if (level == 3) { // Если выбран сложный уровень
        questions = 15; // Задаём 15 вопросов
        maxNumber = 12; // Числа до 12
    } else if (level != 1) { // Если уровень не распознан
        cout << "Неизвестный уровень. Будет выбран лёгкий." << endl; // Предупреждаем пользователя
    } // Конец выбора уровня

    int correct = 0; // Счётчик правильных ответов
    int q = 1; // Номер вопроса
    while (q <= questions) { // Задаём все вопросы
        int a = rand() % maxNumber + 1; // Случайное первое число
        int b = rand() % maxNumber + 1; // Случайное второе число
        int answer; // Ответ пользователя
        cout << q << ") " << a << " * " << b << " = "; // Показываем пример
        cin >> answer; // Считываем ответ
        if (answer == a * b) { // Проверяем ответ
            cout << "Верно!" << endl; // Хвалим за правильный ответ
            correct++; // Увеличиваем количество правильных
        } else { // Если ответ неверный
            cout << "Неверно. Правильный ответ: " << a * b << endl; // Показываем ответ
        } // Конец проверки
        q++; // Переходим к следующему вопросу
    } // Конец цикла вопросов

    cout << "Правильных ответов: " << correct << " из " << questions << endl; // Выводим результат
    double percent = correct * 100.0 / questions; // Считаем процент правильных ответов
    if (percent >= 90) cout << "Оценка: 5" << endl; // 90 процентов и больше — отлично
    else if (percent >= 75) cout << "Оценка: 4" << endl; // 75 процентов и больше — хорошо
    else if (percent >= 50) cout << "Оценка: 3" << endl; // 50 процентов и больше — удовлетворительно
    else cout << "Оценка: 2" << endl; // Иначе неудовлетворительно

    return 0; // Завершаем программу
}
```

ЗАДАНИЕ 13. Ромб из звёздочек заданного размера

``` C++

#include <iostream> // Подключаем ввод и вывод
using namespace std; // Подключаем стандартное пространство имён

int main() { // Начало программы
    int n; // Размер верхней половины ромба
    cout << "Введите размер ромба (например, 5): "; // Просим размер
    cin >> n; // Считываем размер

    if (n <= 0) { // Проверяем размер
        cout << "Размер должен быть положительным." << endl; // Сообщаем об ошибке
        return 0; // Завершаем программу
    } // Конец проверки

    int row = 1; // Начинаем с первой строки
    while (row <= n) { // Печатаем верхнюю половину вместе со средней строкой
        int spaces = n - row; // Количество пробелов перед звёздами
        while (spaces > 0) { // Печатаем нужные пробелы
            cout << " "; // Выводим один пробел
            spaces--; // Уменьшаем количество пробелов
        } // Конец цикла пробелов
        int stars = row * 2 - 1; // Количество звёзд в строке
        while (stars > 0) { // Печатаем звёзды
            cout << "*"; // Выводим звёздочку
            stars--; // Уменьшаем счётчик звёзд
        } // Конец цикла звёзд
        cout << endl; // Переходим на следующую строку
        row++; // Увеличиваем номер строки
    } // Конец верхней половины

    row = n - 1; // Начинаем нижнюю половину без повторения середины
    while (row >= 1) { // Печатаем строки от n-1 до 1
        int spaces = n - row; // Количество пробелов перед звёздами
        while (spaces > 0) { // Печатаем пробелы
            cout << " "; // Выводим пробел
            spaces--; // Уменьшаем счётчик
        } // Конец цикла пробелов
        int stars = row * 2 - 1; // Количество звёзд
        while (stars > 0) { // Печатаем звёзды
            cout << "*"; // Выводим звёздочку
            stars--; // Уменьшаем счётчик
        } // Конец цикла звёзд
        cout << endl; // Переходим на следующую строку
        row--; // Уменьшаем номер строки
    } // Конец нижней половины

    return 0; // Завершаем программу
}
```

Блок-схема MIDDLE++:
Начало - Открыть смену, обнулить статистику -
 Ввод названия: до 3 попыток - название корректно?
Нет - Закрыть смену - Финальная статистика
Да - Ввод цены: до 3 попыток - цена > 0?
Нет - Закрыть смену - Финальная статистика
Да - Ввод количества: до 3 попыток - количество > 0?
Нет - Закрыть смену - Финальная статистика
Да - Промокод  - Базовая сумма -
 Скидка за объём - Скидка за сумму - Скидка промокода - 
 Налог и округление вниз - Доставка - Итог -
 Итог > 1 000 000?
Да - Запрос подтверждения - Подтверждён?
Нет - Отменить заказ, не менять статистику - Следующий заказ
Да / не требуется - Коэффициент и деление -
Печать чека - Обновить статистику -
Добавить ещё? - Да: Следующий заказ
Нет: Вывести статистику - Закрыть смену - Конец
